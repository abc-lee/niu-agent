    36	// PendingAlert represents a pending alert for frontend
    37	type PendingAlert struct {
    38		Content   string `json:"content"`
    39		Timestamp int64  `json:"timestamp"`
    40	}

    42	// ContextConfig represents context window configuration
    43	type ContextConfig struct {
    44		WarningThreshold    float64 `json:"warningThreshold"`    // 预警线百分比 (0.80 = 80%)
    45		TargetThreshold     float64 `json:"targetThreshold"`     // 目标压缩百分比 (0.50 = 50%)
    46		SleepTriggerMinutes int     `json:"sleepTriggerMinutes"` // 睡眠触发时间（分钟）
    47		ContextWindowSize   int     `json:"contextWindowSize"`   // 上下文窗口大小
    48	}
    49	
    50	// DefaultContextConfig returns default context configuration
    51	func DefaultContextConfig() *ContextConfig {
    52		return &ContextConfig{
    53			WarningThreshold:    0.80,
    54			TargetThreshold:     0.50,
    55			SleepTriggerMinutes: 5,
    56			ContextWindowSize:   200000,
    57		}
    58	}
    59	
    60	// LoadContextConfig loads context config from preferences.json
    61	func LoadContextConfig() *ContextConfig {
    62		cfg := DefaultContextConfig()
    63	
    64		// Try to load from preferences.json
    65		homeDir, err := os.UserHomeDir()
    66		if err != nil {
    67			return cfg
    68		}
    69	
    70		prefsPath := filepath.Join(homeDir, ".niu", "preferences.json")
    71		data, err := os.ReadFile(prefsPath)
    72		if err != nil {
    73			return cfg
    74		}
    75	
    76		var prefs struct {
    77			Context *ContextConfig `json:"context"`
    78		}
    79		if err := json.Unmarshal(data, &prefs); err != nil {
    80			return cfg
    81		}
    82	
    83		if prefs.Context != nil {
    84			if prefs.Context.WarningThreshold > 0 {
    85				cfg.WarningThreshold = prefs.Context.WarningThreshold
    86			}
    87			if prefs.Context.TargetThreshold > 0 {
    88				cfg.TargetThreshold = prefs.Context.TargetThreshold
    89			}
    90			if prefs.Context.SleepTriggerMinutes > 0 {
    91				cfg.SleepTriggerMinutes = prefs.Context.SleepTriggerMinutes
    92			}
    93			if prefs.Context.ContextWindowSize > 0 {
    94				cfg.ContextWindowSize = prefs.Context.ContextWindowSize
    95			}
    96		}
    97	
    98		return cfg
    99	}

   107	func AddPendingAlert(content string) {
   108		pendingAlertsMu.Lock()
   109		defer pendingAlertsMu.Unlock()
   110		pendingAlerts = append(pendingAlerts, PendingAlert{
   111			Content:   content,
   112			Timestamp: time.Now().Unix(),
   113		})
   114		if len(pendingAlerts) > 10 {
   115			pendingAlerts = pendingAlerts[len(pendingAlerts)-10:]
   116		}
   117	}
   118	
   119	func GetAndClearPendingAlerts() []PendingAlert {
   120		pendingAlertsMu.Lock()
   121		defer pendingAlertsMu.Unlock()
   122		alerts := pendingAlerts
   123		pendingAlerts = nil
   124		return alerts
   125	}

   127	// detectPython finds a suitable Python executable for MCP servers
   128	func detectPython() string {
   129		// Get executable directory for relative paths
   130		exePath, _ := os.Executable()
   131		exeDir := filepath.Dir(exePath)
   132	
   133		// Priority: bundled python > venv > system Python
   134		candidates := []string{
   135			// Bundled Python (for packaged release)
   136			filepath.Join(exeDir, "python", "Scripts", "python.exe"),
   137			filepath.Join(".", "python", "Scripts", "python.exe"),
   138			// Development venv
   139			"E:/opencode/venv/Scripts/python.exe",
   140			// User's virtualenv
   141			filepath.Join(exeDir, ".venv", "Scripts", "python.exe"),
   142			// System Python
   143			"C:/Python311/python.exe",
   144			"C:/Python310/python.exe",
   145			// PATH fallback
   146			"python",
   147			"python3",
   148		}
   149	
   150		for _, candidate := range candidates {
   151			cmd := exec.Command(candidate, "--version")
   152			if err := cmd.Run(); err == nil {
   153				if absPath, err := filepath.Abs(candidate); err == nil {
   154					return absPath
   155				}
   156				return candidate
   157			}
   158		}
   159	
   160		slog.Warn("No Python found, using 'python' as fallback")
   161		return "python"
   162	}

   164	// loadMemory loads user memory from ~/.niu/memory.json
   165	func loadMemory() map[string]any {
   166		homeDir, err := os.UserHomeDir()
   167		if err != nil {
   168			slog.Warn("Failed to get home directory", "error", err)
   169			return nil
   170		}
   171	
   172		memoryPath := filepath.Join(homeDir, ".niu", "memory.json")
   173		data, err := os.ReadFile(memoryPath)
   174		if err != nil {
   175			slog.Info("No memory file found, will create on first run")
   176			return nil
   177		}
   178	
   179		var memory map[string]any
   180		if err := json.Unmarshal(data, &memory); err != nil {
   181			slog.Warn("Failed to parse memory file", "error", err)
   182			return nil
   183		}
   184	
   185		return memory
   186	}

   188	// formatMemoryForPrompt formats memory for system prompt injection
   189	func formatMemoryForPrompt(memory map[string]any) string {
   190		var sb strings.Builder
   191		sb.WriteString("\n\n# 我的重要记忆\n\n")
   192	
   193		// Identity
   194		if identity, ok := memory["identity"].(map[string]any); ok {
   195			sb.WriteString("## 我的身份\n\n")
   196			if name, ok := identity["name"].(string); ok && name != "" {
   197				sb.WriteString(fmt.Sprintf("我的名字是 %s。\n", name))
   198			}
   199			if personality, ok := identity["personality"].([]any); ok && len(personality) > 0 {
   200				var traits []string
   201				for _, t := range personality {
   202					if s, ok := t.(string); ok {
   203						traits = append(traits, s)
   204					}
   205				}
   206				if len(traits) > 0 {
   207					sb.WriteString(fmt.Sprintf("我的性格：%s。\n", strings.Join(traits, "、")))
   208				}
   209			}
   210			sb.WriteString("\n")
   211		}
   212	
   213		// Workspace
   214		if workspace, ok := memory["workspace"].(map[string]any); ok {
   215			if path, ok := workspace["path"].(string); ok && path != "" {
   216				sb.WriteString("## 工作目录\n\n")
   217				sb.WriteString(fmt.Sprintf("我的知识库存储在：%s\n\n", path))
   218			}
   219		}
   220	
   221		// User
   222		if user, ok := memory["user"].(map[string]any); ok {
   223			if name, ok := user["name"].(string); ok && name != "" {
   224				sb.WriteString("## 用户信息\n\n")
   225				sb.WriteString(fmt.Sprintf("用户称呼：%s\n", name))
   226			}
   227			if prefs, ok := user["preferences"].([]any); ok && len(prefs) > 0 {
   228				var prefStrs []string
   229				for _, p := range prefs {
   230					if s, ok := p.(string); ok {
   231						prefStrs = append(prefStrs, s)
   232					}
   233				}
   234				if len(prefStrs) > 0 {
   235					sb.WriteString(fmt.Sprintf("用户偏好：%s\n", strings.Join(prefStrs, "、")))
   236					// 明确指示文件处理时的行为
   237					for _, pref := range prefStrs {
   238						if strings.Contains(pref, "年份") || strings.Contains(pref, "year") {
   239							sb.WriteString("\n**重要**：处理文件时，必须先创建年份目录（如 documents/2025/），再将文件复制到该目录。\n")
   240						}
   241					}
   242				}
   243			}
   244		}
   245	
   246		// Add configuration file paths info (relative paths)
   247		homeDir, _ := os.UserHomeDir()
   248	
   249		sb.WriteString("\n## 配置文件位置\n\n")
   250		sb.WriteString("配置文件（相对于程序目录）：\n")
   251		sb.WriteString("- 主配置（LLM）：`config/user-config.json`\n")
   252		sb.WriteString("- LLM预设：`config/llm-presets.json`\n")
   253		sb.WriteString("- Agent定义：`config/agents/niu.md`\n")
   254		sb.WriteString("- MCP服务器：`config/mcp-servers.yaml`\n")
   255		if homeDir != "" {
   256			sb.WriteString(fmt.Sprintf("- 用户记忆：`%s/.niu/memory.json`\n", homeDir))
   257		}
   258	
   259		// First run warning
   260		if firstRun, ok := memory["firstRun"].(bool); ok && firstRun {
   261			sb.WriteString("\n**重要**：这是首次运行，请先询问用户设置工作目录。\n")
   262		}
   263	
   264		return sb.String()
   265	}

   274	func main() {
   275		// Handle _exec mode for supervised subprocesses
   276		if len(os.Args) > 2 && os.Args[1] == "_exec" {
   277			if err := supervise.Daemon(); err != nil {
   278				os.Exit(1)
   279			}
   280			return
   281		}
   282	
   283		// Parse flags
   284		configPath := flag.String("config", "./config", "path to configuration directory")
   285		flag.Parse()
   286	
   287		ctx, cancel := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
   288		defer cancel()
   289	
   290		slog.Info("Niu - Personal Knowledge Assistant starting...")
   291	
   292		// If --settings flag, just open settings and exit
   293		if *showSettings {
   294			if err := launchWindow("settings"); err != nil {
   295				slog.Error("Failed to launch settings window", "error", err)
   296				os.Exit(1)
   297			}
   298			return
   299		}
   300	
   301		// If --graph flag, just open graph and exit
   302		if *showGraph {
   303			if err := launchWindow("graph"); err != nil {
   304				slog.Error("Failed to launch graph window", "error", err)
   305				os.Exit(1)
   306			}
   307			return
   308		}
   309	
   310		// Check if LLM is configured
   311		userConfig, err := assistant.LoadConfig()
   312		if err != nil {
   313			slog.Error("Failed to load config", "error", err)
   314			os.Exit(1)
   315		}
   316	
   317		// Load context configuration
   318		contextConfig = LoadContextConfig()
   319		slog.Info("Context configuration loaded",
   320			"warningThreshold", contextConfig.WarningThreshold,
   321			"targetThreshold", contextConfig.TargetThreshold,
   322			"contextWindowSize", contextConfig.ContextWindowSize,
   323		)
   324	
   325		// Initialize LLM interaction logger
   326		if err := goai.InitLLMLogger("logs"); err != nil {
   327			slog.Error("Failed to initialize LLM logger", "error", err)
   328			// Non-fatal, continue without logging
   329		} else {
   330			slog.Info("LLM interaction logging enabled", "directory", "logs")
   331		}
   332	
   333		// If not configured, show settings first
   334		if assistant.NeedsConfiguration(userConfig) {
   335			slog.Info("LLM configuration required, launching settings window...")
   336			fmt.Println("\n⚠️  首次使用需要配置大模型")
   337			fmt.Println("正在打开设置窗口...\n")
   338	
   339			if err := launchWindow("settings"); err != nil {
   340				slog.Error("Failed to launch settings window", "error", err)
   341				fmt.Println("\n请手动运行: cd ui/settings && npm start")
   342				os.Exit(1)
   343			}
   344	
   345			// Wait a moment for config file to be written
   346			time.Sleep(500 * time.Millisecond)
   347	
   348			// Reload config
   349			userConfig, err = assistant.LoadConfig()
   350			if err != nil {
   351				slog.Error("Failed to reload config", "error", err)
   352				os.Exit(1)
   353			}
   354		}
   355	
   356		// Build LLM config from user settings
   357		llmConfig := llm.Config{
   358			DefaultModel: userConfig.LLM.Model,
   359			APIType:      userConfig.LLM.Type,
   360		}
   361	
   362		if userConfig.LLM.Type == "anthropic" {
   363			llmConfig.Anthropic.APIKey = userConfig.LLM.APIKey
   364			llmConfig.Anthropic.BaseURL = userConfig.LLM.APIBase
   365		} else {
   366			llmConfig.Responses.APIKey = userConfig.LLM.APIKey
   367			llmConfig.Responses.BaseURL = userConfig.LLM.APIBase
   368			llmConfig.Responses.ChatCompletionAPI = true
   369		}
   370	
   371		// Detect Python path for MCP servers
   372		pythonPath := detectPython()
   373		slog.Info("Using Python path", "path", pythonPath)
   374		os.Setenv("PYTHON_PATH", pythonPath)
   375	
   376		// Create Nanobot runtime with full Agent framework
   377		rt, err := nanobotRuntime.NewRuntime(llmConfig, nanobotRuntime.Options{
   378			ConfigDir: *configPath,
   379		})
   380		if err != nil {
   381			slog.Error("Failed to create runtime", "error", err)
   382			os.Exit(1)
   383		}
   384	
   385		// Load config from agent definition directory
   386		slog.Info("Loading config from", "path", *configPath)
   387		cfg, _, err := config.Load(ctx, *configPath, true)
   388		if err != nil {
   389			slog.Error("Failed to read config, using defaults", "error", err)
   390			cfg = &types.Config{
   391				Agents: map[string]types.Agent{
   392					"main": {
   393						HookAgent: types.HookAgent{
   394							Name:  "小妞",
   395							Model: userConfig.LLM.Model,
   396						},
   397					},
   398				},
   399			}
   400		}
   401	
   402		// Debug: log loaded config
   403		slog.Info("Config loaded", "agents", len(cfg.Agents), "entrypoints", cfg.Publish.Entrypoint)
   404		for name, agent := range cfg.Agents {
   405			slog.Info("Agent", "name", name, "agentName", agent.Name, "hasInstructions", agent.Instructions.IsSet(), "mcpServers", agent.MCPServers, "tools", agent.Tools, "agents", agent.Agents)
   406		}
   407		slog.Info("MCP Servers", "count", len(cfg.MCPServers))
   408		for name, server := range cfg.MCPServers {
   409			slog.Info("MCP Server", "name", name, "command", server.Command, "args", server.Args, "cwd", server.Cwd, "workdir", server.Workdir)
   410		}
   411	
   412		// Load user memory and inject into agent instructions
   413		memory := loadMemory()
   414		memoryPrompt := formatMemoryForPrompt(memory)
   415		if memoryPrompt != "" {
   416			slog.Info("Memory loaded, injecting into agent instructions")
   417			// Inject memory into niu agent instructions
   418			if niuAgent, ok := cfg.Agents["niu"]; ok {
   419				if niuAgent.Instructions.IsSet() {
   420					baseInstructions := niuAgent.Instructions.Instructions
   421					niuAgent.Instructions.Instructions = baseInstructions + memoryPrompt
   422	
   423					// Override agent name from memory if set
   424					if identity, ok := memory["identity"].(map[string]any); ok {
   425						if name, ok := identity["name"].(string); ok && name != "" {
   426							niuAgent.Name = name
   427							slog.Info("Agent name overridden from memory", "name", name)
   428						}
   429					}
   430	
   431					cfg.Agents["niu"] = niuAgent
   432					slog.Info("Memory injected into niu agent")
   433				}
   434			}
   435		}
   436	
   437		// Create context with config
   438		ctx = rt.WithTempSession(ctx, cfg)
   439	
   440		slog.Info("Nanobot runtime initialized", "model", userConfig.LLM.Model)
   441	
   442		// CLI mode
   443		if *cliMode {
   444			runCLI(ctx, rt, cfg)
   445			return
   446		}
   447	
   448		// GUI mode - start HTTP server for UI
   449		slog.Info("Starting GUI mode...")
   450		fmt.Println("\n🌟 小妞启动中...")
   451	
   452		// Start HTTP server in background
   453		go func() {
   454			if err := startHTTPServer(ctx, cancel, rt, *port, cfg, pythonPath, userConfig); err != nil {
   455				slog.Error("HTTP server error", "error", err)
   456			}
   457		}()
   458	
   459		slog.Info("API server started", "port", *port)
   460		fmt.Printf("✅ 后端服务已启动 (http://127.0.0.1:%d)\n", *port)
   461	
   462		// Launch floating assistant window
   463		if err := launchWindow("assistant"); err != nil {
   464			slog.Error("Failed to launch assistant window", "error", err)
   465			fmt.Println("\n请手动运行: cd ui/assistant && npm start")
   466			os.Exit(1)
   467		}
   468	}

   470	func runCLI(ctx context.Context, rt *nanobotRuntime.Runtime, cfg *types.Config) {
   471		// Simple REPL for CLI mode
   472		fmt.Println("\n🤖 小妞已就绪！输入你的问题，或输入 'quit' 退出。")
   473		fmt.Println("支持：直接提问、工具调用\n")
   474	
   475		// Find the entrypoint agent
   476		var agentName string
   477		if len(cfg.Publish.Entrypoint) > 0 {
   478			agentName = cfg.Publish.Entrypoint[0]
   479		} else {
   480			for name := range cfg.Agents {
   481				agentName = name
   482				break
   483			}
   484		}
   485	
   486		for {
   487			fmt.Print("\n你: ")
   488			var input string
   489			if _, err := fmt.Scanln(&input); err != nil {
   490				break
   491			}
   492	
   493			if input == "quit" || input == "exit" || input == "q" {
   494				fmt.Println("\n再见！👋")
   495				break
   496			}
   497	
   498			if input == "" {
   499				continue
   500			}
   501	
   502			// Create session context
   503			session := mcp.NewEmptySession(ctx)
   504			session.Set(types.ConfigSessionKey, cfg)
   505			callCtx := mcp.WithSession(types.WithConfig(ctx, *cfg), session)
   506	
   507			// Use CallFromCLI with the agent name
   508			result, err := rt.CallFromCLI(callCtx, agentName, input)
   509			if err != nil {
   510				fmt.Printf("\n错误: %v\n", err)
   511				continue
   512			}
   513	
   514			// Print result
   515			for _, content := range result.Content {
   516				if content.Type == "text" {
   517					fmt.Printf("\n小妞: %s\n", content.Text)
   518				}
   519			}
   520		}
   521	}

   523	func startHTTPServer(ctx context.Context, cancel context.CancelFunc, rt *nanobotRuntime.Runtime, port int, cfg *types.Config, pythonPath string, userConfig *assistant.UserConfig) error {
   524		// Load memory to get workspace path
   525		memory := loadMemory()
   526		workspacePath := ""
   527		if memory != nil {
   528			if ws, ok := memory["workspace"].(map[string]any); ok {
   529				if path, ok := ws["path"].(string); ok {
   530					workspacePath = path
   531				}
   532			}
   533		}
   534	
   535		// Get project root directory (absolute path)
   536		// Use the directory of the executable as project root
   537		execPath, err := os.Executable()
   538		slog.Info("Executable path", "path", execPath, "error", err)
   539	
   540		projectRoot := ""
   541		if err == nil {
   542			projectRoot = filepath.Dir(execPath)
   543			slog.Info("Project root directory", "path", projectRoot)
   544		} else {
   545			slog.Warn("Failed to get executable path", "error", err)
   546		}
   547	
   548		// Set workspace path globally for MCP servers (they read from os.Environ())
   549		// Each MCP server will use WORKSPACE_PATH to determine its own database path:
   550		// - kg-server: workspace/knowledge.db (Kuzu)
   551		// - vector-store: workspace/vectors.db (SQLite)
   552		if workspacePath != "" {
   553			os.Setenv("WORKSPACE_PATH", workspacePath)
   554			slog.Info("Workspace configured globally for MCP servers", "path", workspacePath)
   555		}
   556	
   557		// Create environment provider with workspace path
   558		envProvider := func() (map[string]string, error) {
   559			env := map[string]string{}
   560			for _, kv := range os.Environ() {
   561				k, v, _ := kv, kv[len(kv)-len(kv):], ""
   562				for i, c := range kv {
   563					if c == '=' {
   564						k = kv[:i]
   565						v = kv[i+1:]
   566						break
   567					}
   568				}
   569				env[k] = v
   570			}
   571			env["NIU_MCP"] = "true"
   572			env["PYTHON_PATH"] = pythonPath
   573			// Pass workspace path to MCP servers
   574			// Each server decides its own database name based on WORKSPACE_PATH
   575			if workspacePath != "" {
   576				env["WORKSPACE_PATH"] = workspacePath
   577				slog.Info("Workspace configured for MCP servers", "path", workspacePath)
   578			}

   822		// ========== Chat Session API (with persistence) ==========
   823		mux.HandleFunc("/api/chat/session", func(w http.ResponseWriter, r *http.Request) {
   824			if r.Method != "POST" {

   845				sessionID = uuid.String()
   846			}
   847	
   848			slog.Info("Chat session request", "sessionId", sessionID, "msg_len", len(req.Message))
   849	
   850			// Load conversation history from database
   851			var inputMessages []types.Message
   852			if storedMessages, err := sessionManager.DB.GetRecentMessages(r.Context(), sessionID, 50); err == nil {
   853				// GetRecentMessages 返回正序（最旧在前），直接使用
   854				for _, msg := range storedMessages {
   855					inputMessages = append(inputMessages, types.Message{
   856						ID:   uuid.String(),

   858						Items: []types.CompletionItem{

   860								Content: &mcp.Content{

   865						},
   866					})

   872				ID:   uuid.String(),
   873				Role: "user",
   874				Items: []types.CompletionItem{
   875					{
   876						Content: &mcp.Content{
   877							Type: "text",
   878							Text: req.Message,
   879						},
   880					},
   881				},

   884			// Store user message

   929					if json.Unmarshal([]byte(l1JSON), &eventList) == nil && len(eventList) > 0 {
   930						historyRefContent.WriteString(fmt.Sprintf("## 相关记忆（共 %d 条）\n", len(eventList)))
   931						for i, event := range eventList {

  1004					tidyCtx := context.Background()

  1006						SessionID string `json:"session_id"`
  1007					}{SessionID: sid}

  1010					tidyHTTPReq, _ := http.NewRequestWithContext(tidyCtx, "POST",
  1011						"http://localhost:8080/api/context/tidy",
  1012						bytes.NewReader(tidyReqBody))
  1013					tidyHTTPReq.Header.Set("Content-Type", "application/json")

  1016					resp, err := client.Do(tidyHTTPReq)
  1017					if err != nil {
  1018						slog.Error("Async tidy failed", "error", err)
  1019						return
  1020					}
  1021					defer resp.Body.Close()
  1022	
  1023					var tidyResult map[string]any
  1024					if json.NewDecoder(resp.Body).Decode(&tidyResult) == nil {
  1025						slog.Info("Async tidy completed", "result", tidyResult)
  1026					}
  1027				}(sessionID)
  1028			} else {
  1029				slog.Info("Context window: using full history",
  1030					"estimated_tokens", estimatedTokens,
  1031					"message_count", len(inputMessages))

  1033	
  1034			// 构建 LLM 客户端
  1035			llmConfig := llm.Config{
  1036				DefaultModel: userConfig.LLM.Model,
  1037				APIType:      userConfig.LLM.Type,
  1038			}

  1068			if err != nil {

  1072			}

  1076			if resp != nil && len(resp.Output.Items) > 0 {
  1077				for _, item := range resp.Output.Items {
  1078					if item.Content != nil && item.Content.Type == "text" {
  1079						reply += item.Content.Text

  1081				}
  1082			}
  1083	
  1084			// Store assistant reply
  1085			assistantMsg := &session.Message{

  1087				Role:      "assistant",

  1097			if msgCount > 0 && msgCount%20 == 0 { // 20 = 10轮对话（每轮2条消息）

  1099				// 异步调用整理 API
  1100				go func() {

  1596									"type":  "l1",
  1597									"l2_id": l2ID,
  1598								},
  1599							})
  1600							if storeErr == nil {
  1601								l1GeneratedCount++
  1602							}
  1603						}
  1604					}
  1605					if l1GeneratedCount > 0 {
  1606						slog.Info("Generated L1 summaries from L2", "count", l1GeneratedCount)
  1607					}
  1608				}
  1609			}
  1610	
  1611			// 2. 删除冗余消息（保护最近消息）
  1612			deletedCount := 0
  1613			deleteIndices := tidyResponse.ToDelete
  1614			if len(deleteIndices) > 0 {
  1615				// 保护最近 10 条消息不被删除
  1616				recentProtectCount := 10
  1617				totalMessages := len(messages)
  1618				minKeepIndex := totalMessages - recentProtectCount
  1619				if minKeepIndex < 0 {
  1620					minKeepIndex = 0
  1621				}
  1622	
  1623				// 从后往前删除，避免索引变化
  1624				sortedIndices := make([]int, len(deleteIndices))
  1625				copy(sortedIndices, deleteIndices)
  1626				for i, j := 0, len(sortedIndices)-1; i < j; i, j = i+1, j-1 {
  1627					sortedIndices[i], sortedIndices[j] = sortedIndices[j], sortedIndices[i]
  1628				}
  1629	
  1630				var idsToDelete []string
  1631				protectedCount := 0
  1632				for _, idx := range sortedIndices {
  1633					if idx >= 0 && idx < len(messages) {
  1634						// 保护最近的消息
  1635						if idx >= minKeepIndex {
  1636							protectedCount++
  1637							continue
  1638						}
  1639						idsToDelete = append(idsToDelete, fmt.Sprintf("%d", messages[idx].ID))
  1640					}
  1641				}
  1642	
  1643				if protectedCount > 0 {
  1644					slog.Info("Protected recent messages from deletion", "count", protectedCount)
  1645				}
  1646	
  1647				if len(idsToDelete) > 0 {
  1648					count, _ := sessionManager.DB.DeleteMessagesByIDs(r.Context(), idsToDelete)
  1649					deletedCount = int(count)
  1650				}
  1651			}
  1652	
  1653			// 更新整理状态
  1654			finalMsgCount := currentMsgCount - deletedCount
  1655			tidyState := struct {
  1656				MessageCount int `json:"message_count"`
  1657			}{MessageCount: finalMsgCount}
  1658			if data, err := json.Marshal(tidyState); err == nil {
  1659				os.WriteFile(configPath, data, 0644)
  1660			}
  1661	
  1662			slog.Info("Context tidy completed", "stored", storedCount, "deleted", deletedCount, "remaining", finalMsgCount)
  1663	
  1664			json.NewEncoder(w).Encode(map[string]any{
  1665				"status":        "completed",
  1666				"stored_count":  storedCount,
  1667				"deleted_count": deletedCount,
  1668				"message_count": finalMsgCount,
  1669			})
  1670		})
  1671	
  1672		// ========== 便签 API ==========
  1673

  1676			w.Header().Set("Content-Type", "application/json; charset=utf-8")
  1677			w.Header().Set("Access-Control-Allow-Origin", "*")
  1678			w.Header().Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
  1679			w.Header().Set("Access-Control-Allow-Headers", "Content-Type")
  1680	
  1681			if r.Method == "OPTIONS" {
  1682				w.WriteHeader(http.StatusOK)

  1684			}
  1685	
  1686			if r.Method == "GET" {

  1688				json.NewEncoder(w).Encode(notes)

  1690			}
  1691	
  1692			if r.Method == "POST" {

  1695					http.Error(w, "Invalid JSON", http.StatusBadRequest)

  1697				}
  1698				created := dataStore.CreateNote(note)
  1699				json.NewEncoder(w).Encode(created)
  1700				return

  1702	
  1703			http.Error(w, "Method not allowed", http.StatusMethodNotAllowed)
  1704		})
  1705	
  1706		// Update/Delete note
  1707		mux.HandleFunc("/api/notes/", func(w http.ResponseWriter, r *http.Request) {
  1708			w.Header().Set("Content-Type", "application/json; charset=utf-8")
  1709			w.Header().Set("Access-Control-Allow-Origin", "*")
  1710			w.Header().Set("Access-Control-Allow-Methods", "PUT, DELETE, OPTIONS")
  1711			w.Header().Set("Access-Control-Allow-Headers", "Content-Type")
  1712	
  1713			if r.Method == "OPTIONS" {
  1714				w.WriteHeader(http.StatusOK)

  1716			}
  1717	
  1718			// Extract note ID from path
  1719			noteID := r.URL.Path[len("/api/notes/"):]
